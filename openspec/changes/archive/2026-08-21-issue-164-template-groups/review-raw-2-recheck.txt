    74	    })?;

    78	            path: dir.to_path_buf(),
    79	            source,
    80	        })?;

    82	    }

    84	    Ok(paths)

    87	impl TemplateRegistry {
    88	    pub fn load_from_dir<P: AsRef<FsPath>>(dir: P) -> Result<Self, TemplateRegistryError> {
    89	        let dir = dir.as_ref();
    90	        let mut templates = HashMap::new();
    91	        let mut hashes = HashMap::new();
    92	        let mut seen_paths: HashMap<String, PathBuf> = HashMap::new();
    93	        let mut broken: Vec<BrokenTemplate> = Vec::new();
    94	
    95	        for path in sorted_dir_paths(dir)? {
    96	            let ext = path
    97	                .extension()
    98	                .and_then(|ext| ext.to_str())
    99	                .map(|ext| ext.to_ascii_lowercase());
   100	            if !matches!(ext.as_deref(), Some("yaml") | Some("yml")) {
   101	                continue;
   102	            }
   103	
   104	            let filename = path
   105	                .file_name()
   106	                .map(|n| n.to_string_lossy().into_owned())
   107	                .unwrap_or_else(|| path.display().to_string());
   108	
   109	            let contents = match std::fs::read_to_string(&path) {
   110	                Ok(c) => c,
   111	                Err(source) => {
   112	                    return Err(TemplateRegistryError::Io {
   113	                        path: path.clone(),
   114	                        source,
   115	                    })
   116	                }
   117	            };
   118	
   119	            let template = match parse_template(&contents) {
   120	                Ok(t) => t,
   121	                Err(source) => {
   122	                    let error = TemplateRegistryError::Parse {
   123	                        path: path.clone(),
   124	                        source,
   125	                    }
   126	                    .to_string();
   127	                    tracing::warn!(%error, "skipping broken template");
   128	                    broken.push(BrokenTemplate { filename, error });
   129	                    continue;
   130	                }
   131	            };
   132	
   133	            if let Err(message) = template.validate() {
   134	                let error = TemplateRegistryError::Validation {
   135	                    path: path.clone(),
   136	                    message,
   137	                }
   138	                .to_string();
   139	                tracing::warn!(%error, "skipping broken template");
   140	                broken.push(BrokenTemplate { filename, error });
   141	                continue;
   142	            }
   143	
   144	            // The id is already taken, so this file is refused and the template already accepted
   145	            // for the id stays served. Quarantined like a parse or validation fault rather than
   146	            // fatal: one copy-pasted file must not take the whole service down (#181).
   147	            if let Some(existing_path) = seen_paths.get(&template.id) {
   148	                let error = TemplateRegistryError::DuplicateId {
   149	                    id: template.id.clone(),
   150	                    first: existing_path.clone(),
   151	                    second: path,
   152	                }
   153	                .to_string();
   154	                tracing::warn!(%error, "skipping broken template");
   155	                broken.push(BrokenTemplate { filename, error });
   156	                continue;
   157	            }
   158	
   159	            seen_paths.insert(template.id.clone(), path);
   160	            hashes.insert(
   161	                template.id.clone(),
   162	                hex::encode(Sha256::digest(contents.as_bytes())),
   163	            );
   164	            templates.insert(template.id.clone(), template);
   165	        }
   166	
   167	        Ok(Self {
   168	            templates,
   169	            hashes,
   170	            paths: seen_paths,
   171	            broken,
   172	        })
   173	    }

   236	    pub fn validate_params(&self) -> Result<(), String> {
   237	        for (name, spec) in &self.params {
   238	            validate_param_name(name)?;
   239	            validate_param_spec(name, spec)?;
   240	        }
   241	        Ok(())
   242	    }
   243	
   244	    pub fn validate_references(&self) -> Result<(), String> {
   245	        match &self.format {
   246	            TemplateFormat::Single { width, height, .. } => {
   247	                for (dim_name, dim) in [("width", width), ("height", height)] {
   248	                    match dim {
   249	                        DynamicDimension::Fixed(DynamicValue::Ref(r)) => {
   250	                            check_param_ref(
   251	                                &self.params,
   252	                                r,
   253	                                &format!("format {dim_name}"),
   254	                                &["length", "number", "integer"],
   255	                            )?;
   256	                        }
   257	                        DynamicDimension::Dynamic { min, max } => {
   258	                            if let Some(DynamicValue::Ref(r)) = min {
   259	                                check_param_ref(
   260	                                    &self.params,
   261	                                    r,
   262	                                    &format!("format {dim_name}.min"),
   263	                                    &["length", "number", "integer"],
   264	                                )?;
   265	                            }
   266	                            if let Some(DynamicValue::Ref(r)) = max {
   267	                                check_param_ref(
   268	                                    &self.params,
   269	                                    r,
   270	                                    &format!("format {dim_name}.max"),
   271	                                    &["length", "number", "integer"],
   272	                                )?;
   273	                            }
   274	                        }
   275	                        _ => {}
   276	                    }
   277	                }
   278	            }
   279	            TemplateFormat::Sheet { .. } => {}
   280	        }
   281	
   282	        match &self.layout {
   283	            Layout::Items(items) => {
   284	                for item in items {
   285	                    validate_item_references(item, &self.params)?;
   286	                }
   287	            }
   288	        }
   289	        Ok(())
   290	    }

   292	    pub fn instantiate_with_defaults(&self) -> TemplateDefinition {
   293	        TemplateDefinition {

   295	            name: self.name.clone(),
   296	            description: self.description.clone(),
   297	            unit: self.unit.clone(),
   298	            dpi: self.dpi,
   299	            format: instantiate_format_defaults(&self.format, &self.params),
   300	            params: self.params.clone(),
   301	            layout: instantiate_layout_defaults(&self.layout, &self.params),
   302	            version: self.version.clone(),
   303	        }
   304	    }

   306	    pub fn validate(&self) -> Result<(), String> {
   307	        if self.id.trim().is_empty() {
   308	            return Err("id must not be empty".to_string());
   309	        }
   310	        if self.name.trim().is_empty() {
   311	            return Err("name must not be empty".to_string());
   312	        }
   313	        match self.unit.as_str() {
   314	            "mm" | "in" => {}
   315	            _ => return Err("unit must be either \"mm\" or \"in\"".to_string()),
   316	        }
   317	        if self.dpi == 0 {
   318	            return Err("dpi must be greater than 0".to_string());
   319	        }
   320	
   321	        self.validate_params()?;
   322	        self.validate_references()?;
   323	
   324	        if let Some(options) = &self.options() {
   325	            if options.0.is_empty() {
   326	                return Err("options must not be empty".to_string());
   327	            }
   328	            for (name, values) in &options.0 {
   329	                if name.trim().is_empty() {
   330	                    return Err("options must not contain empty names".to_string());
   331	                }
   332	                if values.is_empty() {
   333	                    return Err(format!("options for '{name}' must not be empty"));
   334	                }
   335	                if values.iter().any(|opt| opt.trim().is_empty()) {
   336	                    return Err("options must not contain empty values".to_string());
   337	                }
   338	            }
   339	        }
   340	
   341	        let instantiated = self.instantiate_with_defaults();
   342	
   343	        // Require both bounds on a dynamic-width single before computing layout bounds,
   344	        // so the caller gets the correct error rather than an out-of-bounds panic.
   345	        if let TemplateFormat::Single {
   346	            width: DynamicDimension::Dynamic { min, max },
   347	            ..
   348	        } = &instantiated.format
   349	        {
   350	            if min.is_none() || max.is_none() {
   351	                return Err(
   352	                    "a dynamic-width single template must specify both width.min and width.max"
   353	                        .to_string(),
   354	                );
   355	            }
   356	        }
   357	
   358	        let bounds = layout_bounds(&instantiated.format)?;
   359	        let is_dynamic_width = matches!(
   360	            &instantiated.format,
   361	            TemplateFormat::Single {
   362	                width: DynamicDimension::Dynamic { .. },
   363	                ..
   364	            }
   365	        );
   366	        validate_layout(
   367	            &instantiated.layout,
   368	            instantiated.options().as_ref(),
   369	            bounds.as_ref(),
   370	            is_dynamic_width,
   371	        )?;
   372	
   373	        if let TemplateFormat::Single {
   374	            media_width: Some(mw),
   375	            ..
   376	        } = &instantiated.format
   377	        {
   378	            if *mw <= 0.0 {
   379	                return Err("media_width must be greater than 0".to_string());
   380	            }
   381	        }
   382	
   383	        match &instantiated.format {
   384	            TemplateFormat::Sheet {
   385	                paper_width,
   386	                paper_height,
   387	                label_width,
   388	                label_height,
   389	                positions,
   390	            } => {
   391	                if *paper_width <= 0.0 {
   392	                    return Err("paper_width must be greater than 0".to_string());
   393	                }
   394	                if *paper_height <= 0.0 {
   395	                    return Err("paper_height must be greater than 0".to_string());
   396	                }
   397	                if *label_width <= 0.0 {
   398	                    return Err("label_width must be greater than 0".to_string());
   399	                }
   400	                if *label_height <= 0.0 {
   401	                    return Err("label_height must be greater than 0".to_string());
   402	                }
   403	                if positions.is_empty() {
   404	                    return Err("positions must not be empty".to_string());
   405	                }
   406	                for (idx, position) in positions.iter().enumerate() {
   407	                    let point = position.point();
   408	                    if point.x < 0.0 || point.y < 0.0 {
   409	                        return Err(format!(
   410	                            "position {} must have non-negative coordinates",
   411	                            idx
   412	                        ));
   413	                    }
   414	                }
   415	            }
   416	            TemplateFormat::Single { width, height, .. } => {
   417	                validate_dimension("width", width)?;
   418	                validate_dimension("height", height)?;
   419	            }
   420	        }
   421	
   422	        Ok(())
   423	    }
   424	}

   426	fn validate_param_name(name: &str) -> Result<(), String> {
   427	    if name.is_empty() {
   428	        return Err("parameter name must not be empty".to_string());
   429	    }
   430	    if name == "datetime" || name == "vars" {
   431	        return Err(format!("parameter name '{name}' is reserved"));
   432	    }
   433	    if name.starts_with("datetime.") || name.starts_with("vars.") {
   434	        return Err(format!("parameter name '{name}' uses a reserved prefix"));
   435	    }
   436	    if !name
   437	        .chars()
   438	        .all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '-')
   439	    {
   440	        return Err(format!(
   441	            "parameter name '{name}' contains invalid characters; must match ^[a-zA-Z0-9_-]+$"
   442	        ));
   443	    }
   444	    Ok(())
   445	}
   446	
   447	fn validate_param_spec(name: &str, spec: &ParamSpec) -> Result<(), String> {
   448	    match &spec.param_type {
   449	        ParamType::Enum { values } => {
   450	            if values.is_empty() {
   451	                return Err(format!("parameter '{name}' enum values must not be empty"));
   452	            }
   453	            if values.iter().any(|opt| opt.trim().is_empty()) {
   454	                return Err("options must not contain empty values".to_string());
   455	            }
   456	            if let Some(default) = &spec.default {
   457	                let default_str = match default {
   458	                    crate::models::ParamValue::String(s) => s.as_str(),
   459	                    crate::models::ParamValue::Integer(i) => {
   460	                        let s = i.to_string();
   461	                        if !values.contains(&s) {
   462	                            return Err(format!(
   463	                                "parameter '{name}' default '{i}' is not in enum values"
   464	                            ));
   465	                        }
   466	                        return Ok(());
   467	                    }
   468	                    _ => "",
   469	                };
   470	                if !values.iter().any(|v| v == default_str) {
   471	                    return Err(format!(
   472	                        "parameter '{name}' default '{default_str}' is not in enum values"
   473	                    ));
   474	                }
   475	            }
   476	        }
   477	        ParamType::Length | ParamType::Number | ParamType::Integer => {
   478	            if let (Some(min), Some(max)) = (spec.min, spec.max) {
   479	                if min > max {
   480	                    return Err(format!(
   481	                        "parameter '{name}' min ({min}) must be <= max ({max})"
   482	                    ));
   483	                }
   484	            }
   485	        }
   486	        _ => {}
   487	    }
   488	    Ok(())
   489	}

   491	fn check_param_ref(
   492	    params: &std::collections::BTreeMap<String, ParamSpec>,
   493	    name: &str,
   494	    context: &str,
   495	    allowed_types: &[&str],
   496	) -> Result<(), String> {
   497	    let spec = params
   498	        .get(name)
   499	        .ok_or_else(|| format!("undeclared parameter '{name}' referenced in {context}"))?;
   500	    let matches_type = match &spec.param_type {
   501	        ParamType::Length => allowed_types.contains(&"length"),
   502	        ParamType::Number => allowed_types.contains(&"number"),
   503	        ParamType::Integer => allowed_types.contains(&"integer"),
   504	        ParamType::String { .. } => allowed_types.contains(&"string"),
   505	        ParamType::Boolean => allowed_types.contains(&"boolean"),
   506	        ParamType::Enum { .. } => allowed_types.contains(&"enum"),
   507	    };
   508	    if !matches_type {
   509	        return Err(format!(
   510	            "parameter '{name}' of type {:?} cannot be used in {context}",
   511	            spec.param_type
   512	        ));
   513	    }
   514	    Ok(())
   515	}

   517	fn validate_when_references(
   518	    when: Option<&std::collections::BTreeMap<String, String>>,
   519	    params: &std::collections::BTreeMap<String, ParamSpec>,
   520	) -> Result<(), String> {
   521	    if let Some(when) = when {
   522	        for (name, val) in when {
   523	            let spec = params.get(name).ok_or_else(|| {
   524	                format!("undeclared parameter '{name}' referenced in when condition")
   525	            })?;
   526	            if let ParamType::Enum { values } = &spec.param_type {
   527	                if !values.iter().any(|v| v == val) {
   528	                    return Err(format!(
   529	                        "when condition for '{name}' references '{val}' which is not in enum values"
   530	                    ));
   531	                }
   532	            }
   533	        }
   534	    }
   535	    Ok(())
   536	}

   538	fn validate_item_references(
   539	    item: &LayoutItem,
   540	    params: &std::collections::BTreeMap<String, ParamSpec>,
   541	) -> Result<(), String> {
   542	    match item {
   543	        LayoutItem::Text {
   544	            placement,
   545	            font_weight,
   546	            when,
   547	            ..
   548	        } => {
   549	            validate_when_references(when.as_ref(), params)?;
   550	            if let Some(DynamicValue::Ref(ref_name)) = font_weight {
   551	                check_param_ref(params, ref_name, "font_weight", &["integer"])?;
   552	            }
   553	            if let Extent::Size(size) = &placement.extent {
   554	                for (axis, sv) in [("width", &size.0[0]), ("height", &size.0[1])] {
   555	                    if let SizeValue::Dynamic(DynamicValue::Ref(ref_name)) = sv {
   556	                        check_param_ref(
   557	                            params,
   558	                            ref_name,
   559	                            &format!("text {axis}"),
   560	                            &["length", "number", "integer"],
   561	                        )?;
   562	                    }
   563	                }
   564	            }
   565	        }
   566	        LayoutItem::Qr {
   567	            placement, when, ..
   568	        } => {
   569	            validate_when_references(when.as_ref(), params)?;
   570	            if let Extent::Size(size) = &placement.extent {
   571	                for (axis, sv) in [("width", &size.0[0]), ("height", &size.0[1])] {
   572	                    if let SizeValue::Dynamic(DynamicValue::Ref(ref_name)) = sv {
   573	                        check_param_ref(
   574	                            params,
   575	                            ref_name,
   576	                            &format!("qr {axis}"),
   577	                            &["length", "number", "integer"],
   578	                        )?;
   579	                    }
   580	                }
   581	            }
   582	        }
   583	        LayoutItem::Image {
   584	            placement, when, ..
   585	        } => {
   586	            validate_when_references(when.as_ref(), params)?;
   587	            if let Extent::Size(size) = &placement.extent {
   588	                for (axis, sv) in [("width", &size.0[0]), ("height", &size.0[1])] {
   589	                    if let SizeValue::Dynamic(DynamicValue::Ref(ref_name)) = sv {
   590	                        check_param_ref(
   591	                            params,
   592	                            ref_name,
   593	                            &format!("image {axis}"),
   594	                            &["length", "number", "integer"],
   595	                        )?;
   596	                    }
   597	                }
   598	            }
   599	        }
   600	        LayoutItem::Line { when, .. } => {
   601	            validate_when_references(when.as_ref(), params)?;
   602	        }
   603	        LayoutItem::Container {
   604	            placement,
   605	            when,
   606	            items,
   607	            ..
   608	        } => {
   609	            validate_when_references(when.as_ref(), params)?;
   610	            if let Extent::Size(size) = &placement.extent {
   611	                for (axis, sv) in [("width", &size.0[0]), ("height", &size.0[1])] {
   612	                    if let SizeValue::Dynamic(DynamicValue::Ref(ref_name)) = sv {
   613	                        check_param_ref(
   614	                            params,
   615	                            ref_name,
   616	                            &format!("container {axis}"),
   617	                            &["length", "number", "integer"],
   618	                        )?;
   619	                    }
   620	                }
   621	            }
   622	            for child in items {
   623	                validate_item_references(child, params)?;
   624	            }
   625	        }
   626	    }
   627	    Ok(())
   628	}

   630	fn resolve_f32_default(params: &std::collections::BTreeMap<String, ParamSpec>, name: &str) -> f32 {
   631	    if let Some(spec) = params.get(name) {
   632	        match &spec.default {
   633	            Some(crate::models::ParamValue::Float(f)) => *f,
   634	            Some(crate::models::ParamValue::Integer(i)) => *i as f32,
   635	            Some(crate::models::ParamValue::String(s)) => {
   636	                s.parse::<f32>().unwrap_or_else(|_| spec.min.unwrap_or(0.0))
   637	            }
   638	            _ => spec.min.unwrap_or(0.0),
   639	        }
   640	    } else {
   641	        0.0
   642	    }
   643	}

   650	            _ => 400,

   657	fn instantiate_format_defaults(
   658	    format: &TemplateFormat,
   659	    params: &std::collections::BTreeMap<String, ParamSpec>,
   660	) -> TemplateFormat {
   661	    match format {
   662	        TemplateFormat::Single {
   663	            width,
   664	            height,
   665	            media_width,
   666	        } => {
   667	            let inst_dim = |dim: &DynamicDimension| -> DynamicDimension {
   668	                match dim {
   669	                    DynamicDimension::Fixed(dv) => {
   670	                        let val = match dv {
   671	                            DynamicValue::Literal(v) => *v,
   672	                            DynamicValue::Ref(r) => resolve_f32_default(params, r),
   673	                        };
   674	                        DynamicDimension::Fixed(DynamicValue::Literal(val))
   675	                    }
   676	                    DynamicDimension::Dynamic { min, max } => {
   677	                        let inst_dv =
   678	                            |dv: &Option<DynamicValue<f32>>| -> Option<DynamicValue<f32>> {
   679	                                dv.as_ref().map(|v| match v {
   680	                                    DynamicValue::Literal(val) => DynamicValue::Literal(*val),
   681	                                    DynamicValue::Ref(r) => {
   682	                                        DynamicValue::Literal(resolve_f32_default(params, r))
   683	                                    }
   684	                                })
   685	                            };
   686	                        DynamicDimension::Dynamic {
   687	                            min: inst_dv(min),
   688	                            max: inst_dv(max),
   689	                        }
   690	                    }
   691	                }
   692	            };
   693	            TemplateFormat::Single {
   694	                width: inst_dim(width),
   695	                height: inst_dim(height),
   696	                media_width: *media_width,
   697	            }
   698	        }
   699	        TemplateFormat::Sheet { .. } => format.clone(),
   700	    }
   701	}

   703	fn instantiate_layout_defaults(
   704	    layout: &Layout,
   705	    params: &std::collections::BTreeMap<String, ParamSpec>,
   706	) -> Layout {
   707	    match layout {
   708	        Layout::Items(items) => Layout::Items(
   709	            items

   711	                .map(|item| instantiate_item_defaults(item, params))
   712	                .collect(),

   717	fn instantiate_item_defaults(
   718	    item: &LayoutItem,
   719	    params: &std::collections::BTreeMap<String, ParamSpec>,
   720	) -> LayoutItem {
   721	    let inst_placement = |placement: &crate::models::Placement| -> crate::models::Placement {
   722	        let extent = match &placement.extent {
   723	            Extent::Size(Size([w, h])) => {
   724	                let inst_sv = |sv: &SizeValue| -> SizeValue {
   725	                    match sv {
   726	                        SizeValue::Dynamic(DynamicValue::Ref(r)) => SizeValue::Dynamic(
   727	                            DynamicValue::Literal(resolve_f32_default(params, r)),
   728	                        ),
   729	                        _ => sv.clone(),
   730	                    }
   731	                };
   732	                Extent::Size(Size([inst_sv(w), inst_sv(h)]))
   733	            }
   734	            Extent::To(pos) => Extent::To(pos.clone()),
   735	        };
   736	        crate::models::Placement {
   737	            at: placement.at.clone(),
   738	            extent,
   739	            max_w: placement.max_w,
   740	            max_h: placement.max_h,
   741	            rotate: placement.rotate,
   742	        }
   743	    };
   744	
   745	    match item {
   746	        LayoutItem::Text {
   747	            value,
   748	            placement,
   749	            font_size,
   750	            font_weight,
   751	            multiline,
   752	            alignment,
   753	            when,
   754	        } => {
   755	            let fw = font_weight.as_ref().map(|w| match w {
   756	                DynamicValue::Literal(v) => DynamicValue::Literal(*v),
   757	                DynamicValue::Ref(r) => DynamicValue::Literal(resolve_u16_default(params, r)),
   758	            });
   759	            LayoutItem::Text {
   760	                value: value.clone(),
   761	                placement: inst_placement(placement),
   762	                font_size: font_size.clone(),
   763	                font_weight: fw,
   764	                multiline: *multiline,
   765	                alignment: alignment.clone(),
   766	                when: when.clone(),
   767	            }
   768	        }
   769	        LayoutItem::Qr {
   770	            value,
   771	            placement,
   772	            params: qr_params,
   773	            when,
   774	        } => LayoutItem::Qr {
   775	            value: value.clone(),
   776	            placement: inst_placement(placement),
   777	            params: qr_params.clone(),
   778	            when: when.clone(),
   779	        },
   780	        LayoutItem::Image {
   781	            name,
   782	            src,
   783	            placement,
   784	            fit,
   785	            when,
   786	        } => LayoutItem::Image {
   787	            name: name.clone(),
   788	            src: src.clone(),
   789	            placement: inst_placement(placement),
   790	            fit: *fit,
   791	            when: when.clone(),
   792	        },
   793	        LayoutItem::Line {
   794	            at,
   795	            to,
   796	            thickness,
   797	            when,
   798	        } => LayoutItem::Line {
   799	            at: at.clone(),
   800	            to: to.clone(),
   801	            thickness: *thickness,
   802	            when: when.clone(),
   803	        },
   804	        LayoutItem::Container {
   805	            placement,
   806	            when,
   807	            frame,
   808	            padding,
   809	            items,
   810	        } => LayoutItem::Container {
   811	            placement: inst_placement(placement),
   812	            when: when.clone(),
   813	            frame: frame.clone(),
   814	            padding: *padding,
   815	            items: items
   816	                .iter()
   817	                .map(|child| instantiate_item_defaults(child, params))
   818	                .collect(),
   819	        },
   820	    }
   821	}

   823	fn validate_dimension(name: &str, dimension: &DynamicDimension) -> Result<(), String> {
   824	    match dimension {
   825	        DynamicDimension::Fixed(DynamicValue::Literal(value)) => {
   826	            if *value <= 0.0 {
   827	                return Err(format!("{name} must be greater than 0"));
   828	            }
   829	        }
   830	        DynamicDimension::Fixed(DynamicValue::Ref(_)) => {}
   831	        DynamicDimension::Dynamic { min, max } => {
   832	            if min.is_none() && max.is_none() {
   833	                return Err(format!("{name} dynamic must specify min, max, or both"));
   834	            }
   835	            if let Some(DynamicValue::Literal(min)) = min {
   836	                if *min <= 0.0 {
   837	                    return Err(format!("min_{name} must be greater than 0"));
   838	                }
   839	            }
   840	            if let Some(DynamicValue::Literal(max)) = max {
   841	                if *max <= 0.0 {
   842	                    return Err(format!("max_{name} must be greater than 0"));
   843	                }
   844	            }
   845	            if let (Some(DynamicValue::Literal(min)), Some(DynamicValue::Literal(max))) = (min, max)
   846	            {
   847	                if min > max {
   848	                    return Err(format!("min_{name} must be <= max_{name}"));
   849	                }
   850	            }
   851	        }
   852	    }
   853	    Ok(())
   854	}

   857	    layout: &Layout,
   858	    options: Option<&Options>,
   859	    bounds: Option<&LayoutBounds>,
   860	    is_dynamic_width: bool,
   861	) -> Result<(), String> {

   867	fn validate_layout_items(
   868	    items: &[LayoutItem],
   869	    bounds: Option<&LayoutBounds>,
   870	    options: Option<&Options>,
   871	    is_dynamic_width: bool,
   872	) -> Result<(), String> {
   873	    let mut seen_names = HashSet::new();
   874	    for item in items {
   875	        if let Some(name) = layout_item_name(item) {
   876	            if name.trim().is_empty() {
   877	                return Err("layout item name must not be empty".to_string());
   878	            }
   879	            if !seen_names.insert(name.to_string()) {
   880	                return Err(format!("duplicate layout item name '{}'", name));
   881	            }
   882	        }
   883	        validate_layout_item(item, bounds, options, is_dynamic_width)?;
   884	    }
   885	    Ok(())
   886	}

   889	    match item {
   890	        LayoutItem::Image { name, .. } => name.as_deref(),
   891	        LayoutItem::Text { .. }
   892	        | LayoutItem::Qr { .. }
   893	        | LayoutItem::Line { .. }
   894	        | LayoutItem::Container { .. } => None,
   895	    }

   897	
   898	fn validate_layout_item(
   899	    item: &LayoutItem,
   900	    layout_bounds: Option<&LayoutBounds>,
   901	    options: Option<&Options>,
   902	    is_dynamic_width: bool,
   903	) -> Result<(), String> {
   904	    match item {
   905	        LayoutItem::Text {
   906	            value,
   907	            placement,
   908	            font_size,
   909	            font_weight,
   910	            ..
   911	        } => {
   912	            if value.trim().is_empty() {
   913	                return Err("text value must not be empty".to_string());
   914	            }
   915	            validate_placement_position(
   916	                &placement.at,
   917	                placement.width_is_frame_dependent(),
   918	                layout_bounds,
   919	                is_dynamic_width,
   920	            )?;
   921	            validate_font_weight(font_weight.as_ref())?;
   922	            validate_rotation(&placement.rotate, false)?;
   923	            // `allow_auto_fill` is always `true` for text: this axis asks whether the item type has
   924	            // a frame to fall back on (it does), not whether this template's width is dynamic. Keying
   925	            // it off `is_dynamic_width` instead was #155: it let validation accept a `max_h` above the
   926	            // frame on the strength of a fallback the render path (keyed off the item's own
   927	            // frame-dependence) did not have.
   928	            let (width, height) = resolve_size(
   929	                &placement.at,
   930	                &placement.extent,
   931	                placement.max_w,
   932	                placement.max_h,
   933	                layout_bounds,
   934	                true,
   935	            )?;
   936	            validate_bounds(&placement.at, width, height, layout_bounds)?;
   937	            validate_font_size(font_size)?;
   938	        }
   939	        LayoutItem::Qr {
   940	            value,
   941	            placement,
   942	            params,
   943	            ..
   944	        } => {
   945	            if value.trim().is_empty() {
   946	                return Err("qr value must not be empty".to_string());
   947	            }
   948	            validate_placement_position(
   949	                &placement.at,
   950	                placement.width_is_frame_dependent(),
   951	                layout_bounds,
   952	                is_dynamic_width,
   953	            )?;
   954	            validate_rotation(&placement.rotate, false)?;
   955	            let (width, height) = resolve_size(
   956	                &placement.at,
   957	                &placement.extent,
   958	                placement.max_w,
   959	                placement.max_h,
   960	                layout_bounds,
   961	                false,
   962	            )?;
   963	            validate_bounds(&placement.at, width, height, layout_bounds)?;
   964	            if let Some(params) = params {
   965	                if let Some(module_size) = params.module_size {
   966	                    if module_size <= 0.0 {
   967	                        return Err("qr module_size must be greater than 0".to_string());
   968	                    }
   969	                }
   970	                if let Some(quiet_zone) = params.quiet_zone {
   971	                    if quiet_zone < 0.0 {
   972	                        return Err("qr quiet_zone must be >= 0".to_string());
   973	                    }
   974	                }
   975	            }
   976	        }
   977	        LayoutItem::Image { placement, .. } => {
   978	            validate_placement_position(
   979	                &placement.at,
   980	                placement.width_is_frame_dependent(),
   981	                layout_bounds,
   982	                is_dynamic_width,
   983	            )?;
   984	            validate_rotation(&placement.rotate, false)?;
   985	            let (width, height) = resolve_size(
   986	                &placement.at,
   987	                &placement.extent,
   988	                placement.max_w,
   989	                placement.max_h,
   990	                layout_bounds,
   991	                false,
   992	            )?;
   993	            validate_bounds(&placement.at, width, height, layout_bounds)?;
   994	        }
   995	        LayoutItem::Line {
   996	            at,
   997	            to,
   998	            thickness,
   999	            when,
  1000	        } => {
  1001	            validate_when(when.as_ref())?;
  1002	            const LINE_EPSILON: f32 = 1.0e-4;
  1003	            if *thickness <= 0.0 {
  1004	                return Err("line thickness must be greater than 0".to_string());
  1005	            }
  1006	            // Endpoints resolve against the frame before any comparison: `-0.0 == 0.0`, so raw
  1007	            // coordinates would reject a full-width divider as zero-length.
  1008	            let (start, end) = match layout_bounds {
  1009	                Some(bounds) => (
  1010	                    Point {
  1011	                        x: resolve_coord(at.x(), bounds.width),
  1012	                        y: resolve_coord(at.y(), bounds.height),
  1013	                    },
  1014	                    Point {
  1015	                        x: resolve_coord(to.x(), bounds.width),
  1016	                        y: resolve_coord(to.y(), bounds.height),
  1017	                    },
  1018	                ),
  1019	                None => (at.point(), to.point()),
  1020	            };
  1021	            // On a dynamic-width single an x resolved from an edge-relative component is
  1022	            // provisional: it was computed against `max`. Compare x only when both endpoints are
  1023	            // comparable, and let the render pass re-check the rest.
  1024	            let x_comparable =
  1025	                !is_dynamic_width || at.x().is_sign_negative() == to.x().is_sign_negative();
  1026	            let same_x = x_comparable && (start.x - end.x).abs() < LINE_EPSILON;
  1027	            let same_y = (start.y - end.y).abs() < LINE_EPSILON;
  1028	            if same_x && same_y {
  1029	                return Err("line start and end must differ".to_string());
  1030	            }
  1031	            if let Some(bounds) = layout_bounds {
  1032	                for point in [start, end] {
  1033	                    // A resolved x below zero means the inset exceeds `max`, which no smaller
  1034	                    // final width can rescue, so it is rejected here even on a dynamic label.
  1035	                    if point.x < -LINE_EPSILON || point.y < -LINE_EPSILON {
  1036	                        return Err("line must fit within layout bounds".to_string());
  1037	                    }
  1038	                    // The upper x bound binds only on a plain endpoint: an edge-relative one
  1039	                    // resolves to `bounds.width + x` with `x <= 0`, so it can never exceed the
  1040	                    // frame here. A plain endpoint past `max` is a constant no final width can
  1041	                    // rescue, so it is rejected at load even on a dynamic label.
  1042	                    if point.x > bounds.width + LINE_EPSILON
  1043	                        || point.y > bounds.height + LINE_EPSILON
  1044	                    {
  1045	                        return Err("line must fit within layout bounds".to_string());
  1046	                    }
  1047	                }
  1048	            }
  1049	        }
  1050	        LayoutItem::Container {
  1051	            placement,
  1052	            when,
  1053	            frame,
  1054	            padding,
  1055	            items,
  1056	        } => {
  1057	            validate_when(when.as_ref())?;
  1058	            validate_placement_position(
  1059	                &placement.at,
  1060	                placement.width_is_frame_dependent(),
  1061	                layout_bounds,
  1062	                is_dynamic_width,
  1063	            )?;
  1064	            validate_rotation(&placement.rotate, true)?;
  1065	            let rotation = placement
  1066	                .rotate
  1067	                .and_then(crate::models::Rotation::from_degrees)
  1068	                .unwrap_or(crate::models::Rotation::R0);
  1069	            if rotation.is_rotated() {
  1070	                // §4.2.1: the inner author canvas must be a compile-time constant. `auto` stays
  1071	                // banned outright (ADR-0036, unchanged); a `to` is only a problem when its width is
  1072	                // frame-dependent, which on a fixed frame it never is.
  1073	                let unresolvable = match &placement.extent {
  1074	                    Extent::Size(size) => size.0[0].is_auto() || size.0[1].is_auto(),
  1075	                    Extent::To(_) => placement.width_is_frame_dependent(),
  1076	                };
  1077	                if unresolvable {
  1078	                    return Err(
  1079	                        "rotated container must have fixed size (no auto or dynamic dimensions)"
  1080	                            .to_string(),
  1081	                    );
  1082	                }
  1083	                if subtree_uses_auto(items) {
  1084	                    return Err("auto size is not allowed inside a rotated container".to_string());
  1085	                }
  1086	            }
  1087	            let (width, height) = resolve_size(
  1088	                &placement.at,
  1089	                &placement.extent,
  1090	                placement.max_w,
  1091	                placement.max_h,
  1092	                layout_bounds,
  1093	                true,
  1094	            )?;
  1095	            validate_bounds(&placement.at, width, height, layout_bounds)?;
  1096	
  1097	            if let Some(frame) = frame {
  1098	                if frame.thickness <= 0.0 {
  1099	                    return Err("container frame thickness must be greater than 0".to_string());
  1100	                }
  1101	            }
  1102	
  1103	            // Author canvas = full box, swapped for 90/270. Padding is author-space, so it is
  1104	            // subtracted from the (swapped) author dimensions. For R0 this is the existing math.
  1105	            let (canvas_w, canvas_h) = if rotation.swaps_axes() {
  1106	                (height, width)
  1107	            } else {
  1108	                (width, height)
  1109	            };
  1110	            let inner_width = canvas_w - padding.left - padding.right;
  1111	            let inner_height = canvas_h - padding.top - padding.bottom;
  1112	            const CONTENT_EPSILON: f32 = 1.0e-4;
  1113	            if inner_width <= CONTENT_EPSILON || inner_height <= CONTENT_EPSILON {
  1114	                return Err("container padding leaves no room for content".to_string());
  1115	            }
  1116	            let container_bounds = layout_bounds_from_size(inner_width, inner_height)?;
  1117	            // Children of a frame-dependent container on a dynamic-width single may also use auto
  1118	            // width; they resolve to the container inner width at render time. A container sized
  1119	            // to the right edge via `to` is exactly as frame-dependent as an `auto` one. Rotated
  1120	            // containers reject a frame-dependent extent entirely (above), so child_dynamic is
  1121	            // false there.
  1122	            let child_dynamic =
  1123	                is_dynamic_width && !rotation.is_rotated() && placement.width_is_frame_dependent();
  1124	            validate_layout_items(items, Some(&container_bounds), options, child_dynamic)?;
  1125	        }
  1126	    }
  1127	    Ok(())
  1128	}
  1129	
  1130	/// Bounds-check a placement's anchor. A sign-negative component is edge-relative (#146) and
  1131	/// resolves against the frame first. On a dynamic-width single the final width is unknown but
  1132	/// bounded by `max`, so an inset larger than `max` is rejected here and a merely provisional
  1133	/// result is deferred to the render pass.
  1134	fn validate_placement_position(
  1135	    at: &Position,
  1136	    frame_dependent_width: bool,
  1137	    bounds: Option<&LayoutBounds>,
  1138	    is_dynamic_width: bool,
  1139	) -> Result<(), String> {
  1140	    const BOUNDS_EPSILON: f32 = 1.0e-4;
  1141	    if at.x().is_sign_negative() && frame_dependent_width && is_dynamic_width {
  1142	        return Err(
  1143	            "an edge-relative x cannot be combined with an auto or edge-relative width on a \
  1144	             dynamic-width template: both would depend on the label width"
  1145	                .to_string(),
  1146	        );
  1147	    }
  1148	    let Some(bounds) = bounds else {
  1149	        return Ok(());
  1150	    };
  1151	    if resolve_coord(at.x(), bounds.width) < -BOUNDS_EPSILON
  1152	        || resolve_coord(at.y(), bounds.height) < -BOUNDS_EPSILON
  1153	    {
  1154	        return Err("at resolves outside the frame".to_string());
  1155	    }
  1156	    Ok(())
  1157	}
  1158	
  1159	/// True if any item in the subtree uses an `auto` width or height. Used to forbid auto sizing
  1160	/// anywhere inside a rotated container (#98), where author-horizontal maps to physical-vertical
  1161	/// and the dynamic-width measurement model does not apply.
  1162	fn subtree_uses_auto(items: &[LayoutItem]) -> bool {
  1163	    items.iter().any(|item| match item {
  1164	        LayoutItem::Text { placement, .. }
  1165	        | LayoutItem::Qr { placement, .. }
  1166	        | LayoutItem::Image { placement, .. } => placement
  1167	            .size_or_auto()
  1168	            .is_some_and(|s| s.0[0].is_auto() || s.0[1].is_auto()),
  1169	        LayoutItem::Container {
  1170	            placement, items, ..
  1171	        } => {
  1172	            placement
  1173	                .size_or_auto()
  1174	                .is_some_and(|s| s.0[0].is_auto() || s.0[1].is_auto())
  1175	                || subtree_uses_auto(items)
  1176	        }
  1177	        LayoutItem::Line { .. } => false,
  1178	    })
  1179	}

  1181	fn validate_rotation(rotate: &Option<f32>, is_container: bool) -> Result<(), String> {
  1182	    if let Some(deg) = rotate {
  1183	        // Rotation is a container-only inner transform (#98); reject it on any other item,
  1184	        // regardless of value (even 0).
  1185	        if !is_container {
  1186	            return Err("rotation is only supported on containers".to_string());
  1187	        }
  1188	        if crate::models::Rotation::from_degrees(*deg).is_none() {
  1189	            return Err("rotate must be a multiple of 90 degrees".to_string());
  1190	        }
  1191	    }
  1192	    Ok(())
  1193	}

  1195	/// `size = to - at`, both corners resolved against the frame first. Both components must be
  1196	/// strictly positive: `to` is the top-right corner of a box whose bottom-left is `at`.
  1197	fn resolve_to_extent(
  1198	    at: &Position,
  1199	    to: &Position,
  1200	    frame_w: f32,
  1201	    frame_h: f32,
  1202	) -> Result<(f32, f32), String> {
  1203	    let width = resolve_coord(to.x(), frame_w) - resolve_coord(at.x(), frame_w);
  1204	    let height = resolve_coord(to.y(), frame_h) - resolve_coord(at.y(), frame_h);
  1205	    if width <= 0.0 || height <= 0.0 {
  1206	        return Err("to must be above and to the right of at".to_string());
  1207	    }
  1208	    Ok((width, height))
  1209	}

  1211	fn resolve_size(
  1212	    at: &Position,
  1213	    extent: &Extent,
  1214	    max_w: Option<f32>,
  1215	    max_h: Option<f32>,
  1216	    layout_bounds: Option<&LayoutBounds>,
  1217	    allow_auto_fill: bool,
  1218	) -> Result<(f32, f32), String> {
  1219	    let size = match extent {
  1220	        Extent::Size(size) => size,
  1221	        Extent::To(to) => {
  1222	            // Every caller resolves against a frame (`layout_bounds` is always `Some`, and
  1223	            // container recursion passes its inner box), so this is an internal invariant, not an
  1224	            // authoring error. Erroring beats silently returning a zero extent if that changes.
  1225	            let Some(layout_bounds) = layout_bounds else {
  1226	                return Err("to cannot be resolved without a frame".to_string());
  1227	            };
  1228	            return resolve_to_extent(at, to, layout_bounds.width, layout_bounds.height);
  1229	        }
  1230	    };
  1231	    if let Some(max_w) = max_w {
  1232	        if max_w <= 0.0 {
  1233	            return Err("max_w must be greater than 0".to_string());
  1234	        }
  1235	    }
  1236	    if let Some(max_h) = max_h {
  1237	        if max_h <= 0.0 {
  1238	            return Err("max_h must be greater than 0".to_string());
  1239	        }
  1240	    }
  1241	    let fallback = if allow_auto_fill {
  1242	        layout_bounds.map(|bounds| {
  1243	            (
  1244	                (bounds.width - resolve_coord(at.x(), bounds.width)).max(0.0),
  1245	                (bounds.height - resolve_coord(at.y(), bounds.height)).max(0.0),
  1246	            )
  1247	        })
  1248	    } else {
  1249	        None
  1250	    };
  1251	    let width = resolve_size_value(&size.0[0], max_w, fallback.map(|value| value.0), "width")?;
  1252	    let height = resolve_size_value(&size.0[1], max_h, fallback.map(|value| value.1), "height")?;
  1253	    Ok((width, height))
  1254	}

  1256	fn validate_when(when: Option<&std::collections::BTreeMap<String, String>>) -> Result<(), String> {
  1257	    if let Some(when) = when {
  1258	        if when.is_empty() {
  1259	            return Err("when must not be empty".to_string());
  1260	        }
  1261	        for (name, value) in when {
  1262	            if name.trim().is_empty() || value.trim().is_empty() {
  1263	                return Err("when must not contain empty values".to_string());
  1264	            }
  1265	        }
  1266	    }
  1267	    Ok(())
  1268	}
  1269	
  1270	fn resolve_size_value(
  1271	    value: &SizeValue,
  1272	    max: Option<f32>,
  1273	    fallback: Option<f32>,
  1274	    label: &str,
  1275	) -> Result<f32, String> {
  1276	    match value {
  1277	        SizeValue::Dynamic(DynamicValue::Literal(value)) => {
  1278	            if *value <= 0.0 {
  1279	                return Err(format!("size {label} must be greater than 0"));
  1280	            }
  1281	            Ok(*value)
  1282	        }
  1283	        SizeValue::Dynamic(DynamicValue::Ref(_)) => {
  1284	            // Unresolved dynamic parameter reference in template load validation
  1285	            Ok(0.0)
  1286	        }
  1287	        SizeValue::Auto(_) => {
  1288	            // `max_*` caps the resolution of `auto`; it does not replace the fallback. A cap
  1289	            // larger than the room available is simply not binding, so the smaller wins.
  1290	            let resolved = match (max, fallback) {
  1291	                (Some(max), Some(fallback)) => max.min(fallback),
  1292	                (Some(max), None) => max,
  1293	                (None, Some(fallback)) => fallback,
  1294	                (None, None) => {
  1295	                    return Err(format!("size {label} is auto but no max_{label} provided"))
  1296	                }
  1297	            };
  1298	            if resolved <= 0.0 {
  1299	                // Blame whichever of `max`/`fallback` is actually `resolved` (the smaller of the
  1300	                // two, mirroring the `.min()` above): a `max_*` that isn't the binding value is
  1301	                // fine even if it happens to be set, and a fallback of `0` (the anchor leaving no
  1302	                // room) is not a `max_*` authoring error.
  1303	                let max_is_binding = match (max, fallback) {
  1304	                    (Some(max), Some(fallback)) => max <= fallback,
  1305	                    (Some(_), None) => true,
  1306	                    (None, _) => false,
  1307	                };
  1308	                return Err(if max_is_binding {
  1309	                    format!("max_{label} must be greater than 0")
  1310	                } else {
  1311	                    format!("no room left for an auto {label} at this anchor")
  1312	                });
  1313	            }
  1314	            Ok(resolved)
  1315	        }
  1316	    }
  1317	}

  1319	/// The x half of this check is frame-independent even when the frame is not: for an edge-relative
  1320	/// `at.x` it reduces to `at.x + width <= 0` once `W` cancels out of `W + at.x + width <= W`, and
  1321	/// `validate_placement_position` guarantees such an item's width is a compile-time constant. So
  1322	/// there is nothing to defer to the render pass on either axis.
  1323	fn validate_bounds(
  1324	    at: &Position,
  1325	    width: f32,
  1326	    height: f32,
  1327	    layout_bounds: Option<&LayoutBounds>,
  1328	) -> Result<(), String> {
  1329	    const BOUNDS_EPSILON: f32 = 1.0e-4;
  1330	    let Some(layout_bounds) = layout_bounds else {
  1331	        return Ok(());
  1332	    };
  1333	    let x = resolve_coord(at.x(), layout_bounds.width);
  1334	    let y = resolve_coord(at.y(), layout_bounds.height);
  1335	    if x + width > layout_bounds.width + BOUNDS_EPSILON
  1336	        || y + height > layout_bounds.height + BOUNDS_EPSILON
  1337	    {
  1338	        return Err("item must fit within layout bounds".to_string());
  1339	    }
  1340	    Ok(())
  1341	}

  1343	/// The wght axis accepts any value in range; the multiple-of-100 rule is a CSS-style convention that
  1344	/// keeps templates predictable. Enforced here as well as in `convert.rs` so a `LayoutItem` built by
  1345	/// any route — including the many built directly in tests — is checked.
  1346	fn validate_font_weight(font_weight: Option<&DynamicValue<u16>>) -> Result<(), String> {
  1347	    match font_weight {
  1348	        Some(DynamicValue::Literal(weight))
  1349	            if !(100..=900).contains(weight) || weight % 100 != 0 =>
  1350	        {
  1351	            Err(format!(
  1352	                "font_weight must be a multiple of 100 between 100 and 900, got {weight}"
  1353	            ))
  1354	        }
  1355	        _ => Ok(()),
  1356	    }
  1357	}

  1359	fn validate_font_size(font_size: &FontSize) -> Result<(), String> {
  1360	    match font_size {
  1361	        FontSize::Fixed(value) => {
  1362	            if *value <= 0.0 {
  1363	                return Err("font_size must be greater than 0".to_string());
  1364	            }
  1365	        }
  1366	        FontSize::Range { min, max } => {
  1367	            if *min <= 0.0 || *max <= 0.0 {
  1368	                return Err("font_size min/max must be greater than 0".to_string());
  1369	            }
  1370	            if min > max {
  1371	                return Err("font_size min must be <= max".to_string());
  1372	            }
  1373	        }
  1374	    }
  1375	    Ok(())
  1376	}

  1444	/// Test-only registry covering the whole `catalog/` tree plus `tests/fixtures/templates/`.
  1445	///
  1446	/// Nothing ships with the binary any more (#137): templates live in `catalog/` and users install
  1447	/// what they want. The suite still needs all of them — sheet format, options, container rotation, QR
  1448	/// layout and interpolation are only covered by catalog entries or the engine-demo fixtures moved out
  1449	/// of `catalog/` in #135. Flattens both trees into a single temp dir because `load_from_dir` takes one
  1450	/// path and does not recurse, and returns that dir so a test's `templates_dir` matches its registry —
  1451	/// the source/save/delete endpoints read YAML off disk, so a registry that disagreed with the dir
  1452	/// would 404 on them.
  1453	#[cfg(test)]
  1454	pub(crate) fn load_all_for_tests() -> (TemplateRegistry, std::path::PathBuf) {
  1455	    use std::sync::atomic::{AtomicU32, Ordering};
  1456	    static SEQ: AtomicU32 = AtomicU32::new(0);
  1457	    let dir = std::env::temp_dir().join(format!(
  1458	        "labeler-templates-{}-{}",
  1459	        std::process::id(),
  1460	        SEQ.fetch_add(1, Ordering::Relaxed)
  1461	    ));
  1462	    std::fs::create_dir_all(&dir).expect("create merged template dir");
  1463	    // The catalog is nested (tape/brother, sheet/avery, examples) but the registry — and
  1464	    // {config}/templates, where installs land — is flat, so flatten while copying. Ids are unique
  1465	    // across the tree, enforced by `template_ids_are_unique_and_match_filenames` (#135).
  1466	    fn copy_yaml_into(src: &FsPath, dest: &FsPath) {
  1467	        for entry in std::fs::read_dir(src).unwrap_or_else(|e| panic!("read {src:?}: {e}")) {
  1468	            let path = entry.expect("dir entry").path();
  1469	            // symlink_metadata, not is_dir: a symlinked directory under catalog/ could form a cycle
  1470	            // and recurse forever.
  1471	            let meta = std::fs::symlink_metadata(&path).expect("stat catalog entry");
  1472	            if meta.is_dir() {
  1473	                copy_yaml_into(&path, dest);
  1474	            } else if path.extension().is_some_and(|e| e == "yaml" || e == "yml") {
  1475	                let name = path.file_name().expect("file name");
  1476	                let target = dest.join(name);
  1477	                // Flattening means two catalog files with the same filename in different directories
  1478	                // would silently overwrite each other here, before load_from_dir could ever notice a
  1479	                // duplicate id. Fail loudly instead; the CI gate then explains which files collide.
  1480	                assert!(
  1481	                    !target.exists(),
  1482	                    "two catalog templates share the filename {name:?}; ids must be unique tree-wide"
  1483	                );
  1484	                std::fs::copy(&path, target).expect("copy template");
  1485	            }
  1486	        }
  1487	    }
  1488	    copy_yaml_into(FsPath::new("catalog"), &dir);
  1489	    // The engine demos (QR, multiline, sheet options, rotation, interpolation) are test corpus, not
  1490	    // catalog entries (#135). They flatten into the same dir: ids are unique across both roots,
  1491	    // enforced by `template_ids_are_unique_and_match_filenames`. The files must be copied, not just
  1492	    // registered — this dir becomes `templates_dir`, and GET /templates/{id}/source reads
  1493	    // {templates_dir}/{id}.yaml off disk.
  1494	    copy_yaml_into(FsPath::new("tests/fixtures/templates"), &dir);
  1495	    let registry = TemplateRegistry::load_from_dir(&dir).expect("load templates");
  1496	    (registry, dir)
  1497	}

    21	
    22	**Goals:**
    23	
    24	- Group assignment is a one-click action from the Labels page, on one card or on many, and never
    25	  requires typing YAML.
    26	- A move changes exactly one line of the file and leaves comments, key order, and formatting intact.

    28	- No grouping fault reaches further than the file that carries it. An invalid hand-authored `group:`
    29	  quarantines that one template, exactly as any other validation failure does, and never aborts
    30	  startup or touches another template; and the move endpoint cannot create such a file, because it
    31	  validates before it writes.
    32	
    33	**Non-Goals:**
    34	
    35	- Hierarchy. `Shipping/Pallets` is a name, not a path (decision 2).
    36	- Group entities. There is no create-group or delete-group operation and no place a group is stored
    37	  apart from the templates naming it, so an empty group cannot exist.
    38	- Renaming a group across every template that uses it, and drag-and-drop onto a filter chip. Neither
    39	  is part of #164, and neither is queued: each gets its own issue if and when it is wanted.
    40	- Directory reorganization. `{config}/templates/` stays one flat level.

    42	## Decisions
    43	
    44	### 1. The group is a YAML field on the template, not a directory and not a store row
    45	
    46	Alternatives were directory mapping (`templates/<group>/x.yaml`) and a row in the SQLite store
    47	beside favorites.
    48	
    49	Directory mapping loses on four counts, all in code that exists today. The loader walks one level
    50	(`src/templates.rs:95`) and every downstream identity is a basename, including `BrokenTemplateSummary.filename`
    51	(`src/models.rs:36`) and the duplicate-id tie-break just frozen in `template-registry` ("the one whose
    52	filename sorts first in byte order"); recursion turns all of those into relative paths and reopens a
    53	requirement written days ago. It puts a caller-supplied string on the write path, where
    54	`template_file_path` currently derives the filename from an id restricted to `[A-Za-z0-9_-]`
    55	(`src/api.rs:310`), so a group segment would need its own traversal, separator, and case-collision
    56	guards on an endpoint that writes files. It turns a regroup into mkdir plus cross-directory rename
    57	plus pruning the emptied directory, with a half-failed state the current single atomic rename does
    58	not have. And it puts the group outside the artifact, so a catalog install (`ui/src/pages/Catalog.tsx:127`
    59	POSTs the YAML body) arrives ungrouped.
    60	
    61	Store metadata fails the portability test the same way, and additionally splits template truth across
    62	a file and a database, which is what ADR-0006 exists to prevent.
    63	
    64	The YAML field costs one field in each of the three parse-path files and nothing else. Its one real
    65	loss: reorganizing `/config/templates` by hand with `mv` does not group anything, because the
    66	directory carries no meaning. Accepted knowingly.
    67	
    68	Recorded as **ADR-0062, "A template's group is a YAML field, not its directory"**.
    69	
    70	### 2. One flat level
    71	
    72	`group` is a single name. The service does not split on `/` or any other character, so
    73	`Electronics / Cables` is one group whose name happens to contain a slash.
    74	
    75	Hierarchy was rejected for this change, not forever: it costs a tree control with expansion state in
    76	the Labels view, a decision about whether selecting a parent includes its children, per-segment
    77	validation, and a depth cap, for a set that is currently a few dozen templates. Widening later is a
    78	pure loosening of `validate_group_name`, and no existing file or response has to change, so the
    79	cheap option here does not become the wrong one later. Same ADR-0062.
    80	
    81	### 3. Validation lives in one function, shared by the parse path and the move endpoint
    82	
    83	`validate_group_name(&str) -> Result<String, String>` returns the whitespace-stripped name or a
    84	message naming `group`. It is called from `TemplateDefinition::validate()` (so a bad group quarantines
    85	the file like any other validation failure) and from the move handler before it patches anything (so a
    86	bad name is a `422` with nothing written). One definition, so the file path and the API path cannot
    87	drift.
    88	
    89	The domain model stores the stripped value: `TemplateDefinition.group: Option<String>`, `raw.rs` gains
    90	`group: Option<String>` with `#[serde(default)]`, and `convert.rs` strips on the way through. A
    91	non-string value fails at deserialization with `serde_path_to_error` naming `group`, which is why the
    92	spec says "parse or validation message" rather than naming one.
    93	
    94	Bounds: non-empty after stripping, at most 64 characters, no control characters. 64 is a display
    95	bound, not a storage one: these names sit in a filter row and on cards.

    99	is only a naming preference. The mitigation is in the UI: the move dialog offers the names already in
   100	use, so the normal path picks an existing name instead of retyping it.

   103	
   104	The endpoint replaces one subresource and is idempotent, which is `PUT`, not `PATCH`.
   105	
   106	The patch is textual, never a parse-and-re-emit, which is what keeps it inside ADR-0006's intent

   108	(`existing_template_file`), the handler:

   110	1. reads the stored file;

   112	   ambiguous there, and likewise a file whose root is not a block mapping written one key per line.
   113	   A top-level flow mapping parses fine and has no line to patch or to insert, so reflowing it into
   114	   block style would be exactly the whole-file rewrite this design exists to avoid;
   115	3. finds a top-level `group:` line: a line whose first character is at column 0, that is not a
   116	   comment, and whose key is exactly `group`. Column 0 is what makes a nested `group:` key
   117	   (inside `params`, or a layout item) unreachable, since nested keys are indented;

   168	  for ids at `src/templates.rs:204`), `Ungrouped` only while something is ungrouped. Selection is component state, composed with the existing search box by
   169	  `useMemo`, so both narrow at once. A selection other than `All` also hides the Favorites and
   170	  Recents rows, reusing the `searching` condition already in `ui/src/pages/Templates.tsx:137`: those
   171	  rows are drawn from the whole set, and leaving them on screen under a group filter would show
   172	  cards from the groups the user just filtered out.
   173	- Each card shows its group next to the format badge, and gains a **Move to…** item.
   174	- The move dialog is a text input over a `<datalist>` of the groups in use, plus a "Make ungrouped"
   175	  action. Typing an unused name and confirming creates that group by moving a template into it.
   176	- Multi-select: a checkbox per card and a selection bar reading "N selected · Move to…". A bulk move
   177	  issues one request per template with `Promise.allSettled`, then reports successes and failures
   178	  per template, so a partial failure is visible rather than silent.
   179	- One `useMoveTemplateGroup` mutation invalidating the `templates` query, so the grid and filter row
   180	  both refresh from the server's answer rather than from optimistic local state.
   181	
   182	## Risks / Trade-offs
   183	
   184	- **The patcher corrupts a hand-authored file.** → It never writes text the service cannot parse and
   185	  validate (step 7), the write stays atomic, and the tests assert byte equality outside the patched
   186	  line across the whole `catalog/` tree, which is the real corpus of comment-carrying templates.
   187	- **A move silently overwrites a source edit open in another tab.** The editor PUTs the whole file
   188	  with no version check, so a stale draft saved after a move reverts the group. This hazard exists
   189	  today between two editors; grouping widens it to a second writer. Documented here, and optimistic
   190	  concurrency on template writes is a separate issue rather than a rider on this one.
   191	- **Rollback quarantines grouped templates.** `deny_unknown_fields` means a binary from before this
   192	  change rejects a file carrying `group:`. Per `template-registry` those files are quarantined and
   193	  reported as broken, not fatal, so an older binary starts and serves everything else. Recovery is to
   194	  roll forward, or to delete the `group:` lines. Stated in ADR-0062's consequences.
   195	- **Group names drift by case and spelling.** Only the dialog's list of existing names pushes against
   196	  it. Deliberate: enforcement would mean a registry-wide uniqueness rule for a naming preference.
   197	- **Bulk move is N requests.** At a few dozen templates that is fine; a bulk endpoint is not worth its
   198	  contract until the count justifies it.
   199	
   200	## Migration Plan
   201	
   202	No data migration. Every existing template, request, and response stays valid, and `group` is omitted
   203	from responses when unset, so no consumer sees a shape change. Deployment is the normal image roll
   204	(ADR-0016); rollback is the previous image, with the caveat above.
   205

   207	
   208	None. The three decisions that would have changed the specs or the task breakdown (storage model,
   209	flat versus hierarchical, who rewrites the file) were settled before this document was written.
